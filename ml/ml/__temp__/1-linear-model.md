# Линейная модель и МНК

## 1. Средний эмпирический риск

Для линейной модели с w=[0.5,-0.3] и сеткой x от -1 до 1 с шагом 0.1 сгенерировать y с нормальным шумом N(0,0.1), seed=0. Найти среднюю квадратичную ошибку и сохранить в Q. Вывод не нужен.

### Исходные данные и заготовка

```python
import numpy as np

np.random.seed(0) # псевдослучайные числа образуют одну и ту же последовательность (при каждом запуске)
x = np.arange(-1.0, 1.0, 0.1) # аргумент [-1; 1] с шагом 0,1

size_train = len(x)  # размер выборки
w = [0.5, -0.3]  # коэффициенты модели
model_a = lambda m_x, m_w: (m_w[1] * m_x + m_w[0])  # модель
loss = lambda ax, y: (ax - y) ** 2 # квадратическая функция потерь

y = model_a(x, w) + np.random.normal(0, 0.1, len(x)) # целевые значения
```

### Решение


## 2. Прямая методом наименьших квадратов

На сетке x от -1 до 1 с шагом 0.1 сформировать Y=-5.2+0.7*x+шум N(0,0.1), seed=0. Построить X со строками [1,x_i] и найти w=(X^T X)^(-1)X^T Y. Результат — два коэффициента в w; ничего не печатать.

### Исходные данные и заготовка

```python
import numpy as np

np.random.seed(0) # псевдослучайные числа образуют одну и ту же последовательность (при каждом запуске)
x = np.arange(-1.0, 1.0, 0.1) # аргумент [-1; 1] с шагом 0,1

model_a = lambda xx, ww: (ww[0] + ww[1] * xx) # модель
Y = -5.2 + 0.7 * x + np.random.normal(0, 0.1, len(x)) # вектор целевых значений
```

### Решение

```python
import numpy as np

np.random.seed(0) # псевдослучайные числа образуют одну и ту же последовательность (при каждом запуске)
x = np.arange(-1.0, 1.0, 0.1) # аргумент [-1; 1] с шагом 0,1


model_a = lambda xx, ww: (ww[0] + ww[1] * xx) # модель
Y = -5.2 + 0.7 * x + np.random.normal(0, 0.1, len(x)) # вектор целевых значений

# здесь продолжайте программу
X = np.vstack((np.ones(len(x)), x)).T
w = np.linalg.inv(X.T @ X) @ X.T @ Y
```

## 3. Полиномиальная регрессия третьей степени

При seed=0 и x=np.arange(-1,1,0.1) сформировать Y=sin(5*x)+2*x+нормальный шум со стандартным отклонением 0.1. Для модели третьей степени построить строки [1,x,x**2,x**3] и вычислить четыре коэффициента w аналитическим МНК. Вывод не нужен.

### Исходные данные и заготовка

```python
import numpy as np

np.random.seed(0) # псевдослучайные числа образуют одну и ту же последовательность (при каждом запуске)
x = np.arange(-1.0, 1.0, 0.1) # аргумент [-1; 1] с шагом 0,1

model_a = lambda xx, ww: (ww[0] + ww[1] * xx + ww[2] * xx ** 2 + ww[3] * xx ** 3) # модель
Y = np.sin(x * 5) + 2 * x + np.random.normal(0, 0.1, len(x)) # вектор целевых значений

X = np.array([[1, xx, xx**2, xx**3] for xx in x]) # обучающая выборка для поиска коэффициентов w модели a
```

### Решение

```python
import numpy as np

np.random.seed(0)  # псевдослучайные числа образуют одну и ту же последовательность (при каждом запуске)
x = np.arange(-1.0, 1.0, 0.1)  # аргумент [-1; 1] с шагом 0,1

model_a = lambda xx, ww: (ww[0] + ww[1] * xx + ww[2] * xx ** 2 + ww[3] * xx ** 3)  # модель
Y = np.sin(x * 5) + 2 * x + np.random.normal(0, 0.1, len(x))  # вектор целевых значений

X = np.array([[1, xx, xx**2, xx**3] for xx in x])  # обучающая выборка для поиска коэффициентов w модели a
w = np.linalg.inv(X.transpose() @ X) @ X.transpose() @ Y
```

## 4. Регрессия через максимум правдоподобия

Аппроксимировать f(x)=0.5*x+0.2*x**2-0.05*x**3+0.2*sin(4*x)-2.5 на [-4,6] кубическим полиномом. При независимом гауссовском шуме максимизировать правдоподобие: решить уравнение суммы (y_i-w·x_i)*x_i=0 по данным x_train,y_train. Сохранить w и среднюю квадратичную ошибку Q; не печатать.

### Решение

```python
# @title Решение задачи регрессии с помощью теоремы Байеса
import numpy as np

# Функции
f = lambda x: 0.5 * x + 0.2 * x ** 2 - 0.05 * x ** 3 + 0.2 * np.sin(4 * x) - 2.5
p_prod = lambda X, Y, w, D: - 1 / D * (Y - X @ w) @ X

# Данные
coord_x = np.arange(-4.0, 6.0, 0.1)
x_train = np.array([[_x**i for i in range(4)] for _x in coord_x])
y_train = f(coord_x)

# Параметры
N = 10000
w = np.array([.0, .0, .0, .0])
k = np.array([.1, .001, .0001, .00001])

# Вычисление дисперсии
D = np.var(coord_x) # по двум осям

# Вычисление начального качества:
Q = np.mean((x_train @ w - y_train)**2)
# print('Q начальное:', Q)

for _ in range(N):
    grad = p_prod(x_train, y_train, w, D)
    w -= k * grad
    # if _ % 1000 == 0:
    #     Q = np.mean((x_train @ w - y_train)**2)
    #     print(f'Итерация {_}, w: {w}, Q: {Q}')

# Вычисление конечного качества:
Q = np.mean((x_train @ w - y_train)**2)
# print('Q конечное:', Q)
```
