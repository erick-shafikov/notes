# Кластеризация K-means

## 1. Ллойд с абсолютными расстояниями

Разбить заданную неразмеченную data_x на K кластеров алгоритмом Ллойда, начиная со случайных центров ma из заготовки. Расстояние — среднее абсолютных разностей координат. Сохранить X как список списков точек: X[j] соответствует центру ma[j], порядок точек внутри произвольный. Не печатать.

### Решение

```python
import numpy as np

np.random.seed(0)


def metric(x_i, x_k): 
    return abs(x_i - x_k).sum(axis=1).reshape(-1, 1)

data_x = [(48, 118), (74, 96), (103, 82), (135, 76), (162, 79), (184, 97), (206, 111), (231, 118), (251, 118), (275, 110), (298, 86), (320, 68), (344, 62), (376, 61), (403, 75), (424, 95), (440, 114), (254, 80), (219, 85), (288, 66), (260, 92), (201, 76), (162, 66), (127, 135), (97, 143), (83, 160), (82, 177), (88, 199), (105, 205), (135, 208), (151, 198), (157, 169), (153, 152), (117, 158), (106, 168), (106, 185), (123, 188), (125, 171), (139, 163), (139, 183), (358, 127), (328, 132), (313, 146), (300, 169), (300, 181), (308, 197), (326, 206), (339, 209), (370, 199), (380, 184), (380, 147), (343, 154), (329, 169), (332, 184), (345, 185), (363, 159), (361, 177), (344, 169), (311, 175), (351, 89), (134, 96)]

M = np.mean(data_x, axis=0) # вычисление средних по каждой координате
D = np.var(data_x, axis=0) # вычисление дисперсий по каждой координате
K = 3  # число кластеров
ma = np.array([np.random.normal(M, np.sqrt(D / 10), 2) for n in range(K)]) # начальные центры кластеров

for _ in range(10):
    X = [[] for _ in range(K)] # обнуляем группы кластероов
    
    distances = np.hstack([metric(data_x, ma[i]) for i in range(K)])
    clusters = np.argmin(distances, axis=1)  
    
    for x, n_cluster in zip(data_x, clusters):
        X[n_cluster].append(x)
    
    ma = np.array([np.mean(X[i], axis=0) for i in range(K)]) # новые центроиды
```

T = [[(365, 200), (390, 180), (350, 172), (400, 171)], [(77, 150), (100, 200), (50, 130)], [(250, 100), (170, 88), (280, 102), (230, 108)]]
data_x = [(48, 118), (74, 96), (103, 82), (135, 76), (162, 79), (184, 97), (206, 111), (231, 118), (251, 118), (275, 110), (298, 86), (320, 68), (344, 62), (376, 61), (403, 75), (424, 95), (440, 114), (254, 80), (219, 85), (288, 66), (260, 92), (201, 76), (162, 66), (127, 135), (97, 143), (83, 160), (82, 177), (88, 199), (105, 205), (135, 208), (151, 198), (157, 169), (153, 152), (117, 158), (106, 168), (106, 185), (123, 188), (125, 171), (139, 163), (139, 183), (358, 127), (328, 132), (313, 146), (300, 169), (300, 181), (308, 197), (326, 206), (339, 209), (370, 199), (380, 184), (380, 147), (343, 154), (329, 169), (332, 184), (345, 185), (363, 159), (361, 177), (344, 169), (311, 175), (351, 89), (134, 96)]

ma =  [np.mean(T[i],axis=0) for i in range(K)]
ma_T = np.copy(ma)

while n<10:
    ro = np.array([np.mean((data_x-ma[i])**2,axis=1) for i in range(K)]).T
