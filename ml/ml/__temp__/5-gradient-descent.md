# Градиентный спуск

## 1. Минимизация функции одной переменной

```python
# Условие:
# Найти локальный минимум f(x)=0.5*x+0.2*x**2-0.1*x**3 обычным градиентным спуском. Начало x=-4,
# шаг 0.01, ровно 200 итераций. Итог сохранить в x, без вывода.

import numpy as np


def func(x):
    return 0.5 * x + 0.2 * x ** 2 - 0.1 * x ** 3


def df(x):
    return 0.5 + 0.4*x - 0.3 * x ** 2


coord_x = np.arange(-5.0, 5.0, 0.1) # значения по оси абсцисс
coord_y = func(coord_x) # значения по оси ординат (значения функции)

N = 200
x = -4
lmd = 0.01

for i in range(N + 1):
    x = x - lmd * df(x)
```

## 2. Повторение градиентного спуска в уроке кластеризации

```python
# Условие:
# Задача на повторение / аналог предыдущей.
#
# Повторение обычного градиентного спуска: минимизировать f(x)=0.5*x+0.2*x**2-0.1*x**3, начиная
# с x=-4. Использовать eta=0.01 и ровно 200 шагов x-=eta*f′(x). Итог записать в x, без вывода.

import numpy as np

def func(x):
    return 0.5 * x + 0.2 * x ** 2 - 0.1 * x ** 3

def df(x):
    return 0.5 + 0.4 * x - 0.3 * x ** 2

eta = 0.01
x_current = -4.0
N = 200

for i in range(N):
    x_current = x_current - eta * df(x_current)

x = x_current
```
